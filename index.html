1. Aktuálny stav & Cieľ projektu

Praktický cieľ

AMD Radeon Pro 455 má iba 2 GB GDDR5 VRAM (na desktope s Electron aplikáciami sa zaplní na 85 %). Cieľom je presunúť desktop, Antigravity, Edge WebView2, PowerToys na Intel HD Graphics 530 (zdieľaná 16 GB RAM), aby 2 GB VRAM AMD ostali voľné výlučne pre World of Tanks a Whisper — a voliteľne mať aj úsporný režim (Mód 4) s úplne vypnutou AMD pre úsporu ~6 W.

Bezpečný stav & Záchrana

Všetky experimenty bežia výlučne z USB kľúča D: (REFIND_USB) bez akéhokoľvek zásahu do interného EFI Windows Boot Managera. Odpojením USB a reštartom máš vždy okamžitý 100 % funkčný Windows 11 s R.ID 25.3.1, natívnym rozlíšením 2880×1800 @ 60 Hz a funkčným jasom.

Čo sme už dokázali rozbehať a overiť (Checkpoint hotových vecí) 1) USB rEFInd 0.14.2 s spoof_osx_version 10.15 spoľahlivo odblokuje PCI 00:02.0 (Intel HD 530).
2) Nainštalovaný finálny WHQL ovládač Intel 31.0.101.2115 — Intel HD 530 nabieha v D0, Status: OK, renderuje do systémovej RAM.
3) Nainštalovaný R.ID Polaris-Vega-Navi 25.3.1-R2.5 (kernel 31.0.23013.14001) + náš skript Apply-MacBookPro13-3-Fix.ps1 (10 Apple/Polaris 11 registrových kľúčov) — v normálnom boote znížil spotrebu VRAM z 1 747 MB na 1 223 MB pri funkčnom jase.
4) Fáza 0 kompletne dokončená: máme celý WATCHDOG-20261125-1333.dmp dekódovaný do jednej inštrukcie v amdkmdag.sys aj kompletné dekompilované ACPI tabuľky z oboch bootov.

3. Forenzný nález Fázy 0: Presná príčina AMD Code 43

100 % dokázaný vinník v WATCHDOG-20261125-1333.dmp + amdkmdag.sys Po zavolaní AppleSetOs zlyhá štart AMD ovládača DxgkDdiStartDevice (amdkmdag+0x16F4684) na chybe PP Lib initialization failed s presnou bitmaskou pInstance->PECI.failFlag.pplib_init_failure.failure_bitmask = 0x0000000000000801 (Bit 0 + Bit 11).
Jediná inštrukcia v celom 105 MB ovládači, ktorá nastavuje Bit 11 (0x800), je amdkmdag+0x1718329 vnútri smum_start_smc → [smumgr+0x280] = polaris10_start_smu (štart/handshake Polaris 11 SMU7 mikrokontroléra, ~269 ms timeout)!

Vyvrátené hypotézy (100 % overené z dumpu & diffu)

  • OK VBIOS nechýba: VFCT (60 548 B) je identická v oboch bootoch a amdkmdag v LZNT1 bloku Ev=0x900C0E0D úspešne načítal celých 60 416 B ATOMBIOS VBIOS (113-C9801AP-028, 55 AA 76 E9).
  • OK PCIe / BAR / MSI / D0 sú zdravé: dekomprimovaný PCI Config Space v čase pádu (Ev=0x900C0E14) má PCI_CMD=0x0407 (IO+Mem+BusMaster ON), PMCSR=0x0000 (D0), link Gen3 x8 (0x1083), MSI=0x0081 (0xFEE000D0).
  • OK ACPI sa nemení: všetkých 24 tabuliek (dsdt.dat, ssdt..ssdt12, vfct, dmar, mcfg, xsdt) má 100 % zhodný SHA-256 medzi setos a normal bootom.
  • OK BARy sú identické: 0xB0000000, 0xC0000000, 0x4000, 0x82600000.
  • OK Ostatné PowerPlay vrstvy prešli: PECI, SW_BGM, PHM, CWDDEPM, PSM, PEM všetky úspešne vyčistili svoje chybové bity (0x04..0x400).

5.2 Rozhranie 2: Pre-T2 Retina Indexed I/O GMUX (Renesas R4F2113 @ 0x0700–0x07FE)

Hlavný hardvérový rozdiel oproti t2-gmux-uefi (ktorý používa MMIO 0xFE0B0200). Zdrojový protokol overený z Linuxu (apple-gmux.c) aj z tvojho vlastného ssdt1.dsl (SGMX): 

// Fyzické porty (báza 0x0700):
//   0x07C2..0x07C5 = GMUX_PORT_VALUE (8-bit inb/outb, 32-bit inl/outl little-endian)
//   0x07CC/CD/CE   = Detekčný probe: outb(0xAA, 0x7CC); outb(0x55, 0x7CD); outb(0x00, 0x7CE);
//                    is_indexed = ((inb(0x7CC) | (inb(0x7CD) << 8)) == 0x55AA);
//   0x07D0         = GMUX_PORT_READ  (zápis logického portu pre čítanie + ack read)
//   0x07D4         = GMUX_PORT_WRITE (status bit 0: 0=ready, 1=complete + zápis portu pre write)

static BOOLEAN GmuxWaitReady(EFI_BOOT_SERVICES *BS) {
    for (UINTN i = 0; i < 200; i++) {
        if ((IoRead8(0x7D4) & 0x01) == 0) return TRUE;
        (VOID)IoRead8(0x7D0);
        BS->Stall(100);
    }
    return FALSE;
}
static BOOLEAN GmuxWaitComplete(EFI_BOOT_SERVICES *BS) {
    for (UINTN i = 0; i < 200; i++) {
        if ((IoRead8(0x7D4) & 0x01) != 0) { (VOID)IoRead8(0x7D0); return TRUE; }
        BS->Stall(100);
    }
    return FALSE;
}
BOOLEAN GmuxIdxRead8(EFI_BOOT_SERVICES *BS, UINT8 Port, UINT8 *Out) {
    if (!GmuxWaitReady(BS)) return FALSE;
    IoWrite8(0x7D0, Port);
    if (!GmuxWaitComplete(BS)) return FALSE;
    *Out = IoRead8(0x7C2);
    return TRUE;
}
BOOLEAN GmuxIdxWrite8(EFI_BOOT_SERVICES *BS, UINT8 Port, UINT8 Val) {
    IoWrite8(0x7C2, Val);
    if (!GmuxWaitReady(BS)) return FALSE;
    IoWrite8(0x7D4, Port);
    return GmuxWaitComplete(BS);
}

5.4 Rozhranie 4: ACPI SSDT Patcher + Návrh SSDT_IGPU_MBP133.asl

In-memory patchovanie v UEFI (lib/int_acpi.c) je 100 % kompatibilné s HVCI / VBS (beží pred ExitBootServices, nevyžaduje Test Signing). Pre MacBookPro13,3 stačí upraviť int_acpi.c tak, aby hľadal a premenoval _BCM → XBCM v DSDT (v rozsahu IGPU.DD01, nie SaSsdt) a priložiť tento mimoriadne čistý SSDT využívajúci už existujúce Apple metódy: 

// Návrh SSDT_IGPU_MBP133.asl (pre Mód 5 / Mód 4 na MacBookPro13,3)
// Loader v UEFI premenuje v DSDT: IGPU.DD01._BCM -> XBCM a \_WAK -> XWAK
DefinitionBlock ("", "SSDT", 2, "MBP133", "IGPUBRT", 0x00001000)
{
    External (_SB_.PCI0.IGPU.DD01, DeviceObj)
    External (_SB_.PCI0.IGPU.DD01.XBCM, MethodObj)   // 1 Arg (premenované _BCM v DSDT!)
    External (_SB_.PCI0.PEG0.GFX0.ABCM, MethodObj)   // 1 Arg (už existuje v ssdt1.dsl!)
    External (_SB_.PCI0.PEG0.GFX0.SGMX, MethodObj)   // 3 Args (už existuje v ssdt1.dsl!)
    External (BRTL, FieldUnitObj)
    External (XWAK, MethodObj)                       // 1 Arg (premenované _WAK v DSDT)

    Scope (\_SB.PCI0.IGPU.DD01)
    {
        Method (_BCM, 1, NotSerialized)
        {
            XBCM (Arg0)
            If (CondRefOf (\_SB.PCI0.PEG0.GFX0.ABCM))
            {
                \_SB.PCI0.PEG0.GFX0.ABCM (Arg0)
            }
        }
    }

    Method (_WAK, 1, NotSerialized)
    {
        Local0 = XWAK (Arg0)
        If (CondRefOf (\_SB.PCI0.PEG0.GFX0.SGMX))
        {
            \_SB.PCI0.PEG0.GFX0.SGMX (0x0728, One, One)   // DDC -> IGD
            \_SB.PCI0.PEG0.GFX0.SGMX (0x0710, One, 0x02)  // Panel -> IGD
            \_SB.PCI0.PEG0.GFX0.SGMX (0x0740, One, 0x03)  // External -> DIS
            Sleep (0x0A)
            If (CondRefOf (\_SB.PCI0.PEG0.GFX0.ABCM))
            {
                \_SB.PCI0.PEG0.GFX0.ABCM (BRTL)           // Obnoviť jas po prebudení
            }
        }
        Return (Local0)
    }
}
